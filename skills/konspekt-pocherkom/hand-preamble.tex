% --- линии от руки: каждая чуть кривая, с наклоном, разной длины и толщины
% \handseg{(A)}{(B)}: штрих от A до B — прогиб, дрожание, недолёт/перелёт концов
\newcommand{\handseg}[2]{%
  \path #1 coordinate (hsA) #2 coordinate (hsB);%
  \pgfpointdiff{\pgfpointanchor{hsA}{center}}{\pgfpointanchor{hsB}{center}}%
  \pgfmathsetmacro{\hlL}{veclen(\pgf@x,\pgf@y)}%
  \pgfmathsetmacro{\hlA}{(rnd-0.5)*(0.05*\hlL+3)}\pgfmathsetmacro{\hlB}{(rnd-0.5)*(0.05*\hlL+3)}%
  \pgfmathsetmacro{\hlY}{(rnd-0.5)*(0.035*\hlL+1.5)}%
  \pgfmathsetmacro{\hlW}{0.5+rnd*0.45}\pgfmathsetmacro{\hlS}{(rnd-0.4)*0.06}\pgfmathsetmacro{\hlE}{1+(rnd-0.3)*0.06}%
  \draw[line width=\hlW pt, line cap=round, rounded corners=2pt,
        decorate, decoration={random steps, segment length=9pt, amplitude=0.45pt}]
    ($(hsA)!\hlS!(hsB)$) .. controls ($(hsA)!0.33!(hsB)!\hlA pt!90:(hsB)$) and ($(hsA)!0.66!(hsB)!\hlB pt!90:(hsB)$) .. ($(hsA)!\hlE!(hsB)+(0,\hlY pt)$);}
\newcommand{\handline}[2]{\handseg{(0,0)}{(#1,#2)}}
\makeatother
